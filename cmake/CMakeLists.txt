cmake_minimum_required(VERSION 3.25)

# Deliberately separate from the root SCons entry point and its build tree.
get_filename_component(DCPP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
cmake_path(IS_PREFIX DCPP_ROOT "${CMAKE_BINARY_DIR}" NORMALIZE in_repository)
if(CMAKE_BINARY_DIR STREQUAL DCPP_ROOT OR CMAKE_BINARY_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    message(FATAL_ERROR "Use a separate build directory, e.g. -B ../build/cmake/mingw-release")
endif()
if(in_repository)
    set(allowed_build_root "${DCPP_ROOT}/build/cmake")
    cmake_path(IS_PREFIX allowed_build_root "${CMAKE_BINARY_DIR}" NORMALIZE isolated_build)
    if(NOT isolated_build)
        message(FATAL_ERROR "In-repository CMake builds must be under build/cmake/ to isolate SCons outputs")
    endif()
endif()

project(DCPlusPlus LANGUAGES C CXX RC)
if(NOT WIN32)
    message(FATAL_ERROR "This experimental build currently supports Windows only")
endif()
if(NOT MSVC AND NOT MINGW)
    message(FATAL_ERROR "Select MSVC or MinGW-w64")
endif()
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build configuration" FORCE)
endif()
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES Debug Release CACHE STRING "Build configurations" FORCE)
elseif(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Supported configurations: Debug and Release")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib/$<CONFIG>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
set(CMAKE_PDB_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<CONFIG>")
option(DCPP_PCH "Use precompiled headers for the core and GUI" OFF)
set(DCPP_TEST_COMPONENTS "" CACHE STRING "Opt-in tests: dcpp;dwt;win32;zlib;dwarf")
set(DCPP_BUILD_NUMBER "0" CACHE STRING "Independent CMake build label; does not consume the SCons counter")
if(NOT DCPP_BUILD_NUMBER MATCHES "^[0-9]+$")
    message(FATAL_ERROR "DCPP_BUILD_NUMBER must be a nonnegative integer")
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(DCPP_ARCH x64)
    set(DCPP_LIB_SUFFIX /x64)
else()
    set(DCPP_ARCH x86)
    set(DCPP_LIB_SUFFIX "")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include("${CMAKE_CURRENT_SOURCE_DIR}/Compiler.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/Dependencies.cmake")
include("${CMAKE_CURRENT_SOURCE_DIR}/Application.cmake")
if(DCPP_TEST_COMPONENTS)
    enable_testing()
    add_subdirectory("${DCPP_ROOT}/test" "${CMAKE_BINARY_DIR}/test")
endif()
set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT DCPlusPlus)
